// ── Starfield Engine ────────────────────────────────────────────────────── const canvas = document.getElementById('starfield'); const ctx = canvas.getContext('2d'); let width, height; let stars = []; let comets = []; let mx = -1000, my = -1000; function resize() { width = canvas.width = window.innerWidth; height = canvas.height = window.innerHeight; } window.addEventListener('resize', resize); window.addEventListener('mousemove', e => { mx = e.clientX; my = e.clientY; }); window.addEventListener('mouseout', () => { mx = -1000; my = -1000; }); resize(); class Star { constructor() { this.x = Math.random() * width; this.y = Math.random() * height; this.baseSize = Math.random() * 1.3 + 0.5; this.isDiamond = Math.random() < 0.2; this.size = this.isDiamond ? this.baseSize * 1.5 : this.baseSize; this.alpha = Math.random(); this.alphaSpeed = (Math.random() * 0.02 + 0.005) * (Math.random() < 0.5 ? 1 : -1); this.vx = (Math.random() - 0.5) * 0.05; this.vy = -(Math.random() * 0.06 + 0.03); const colors = ['255,255,255', '200,220,255', '255,240,200', '230,200,255', '200,255,230']; this.color = this.isDiamond ? colors[Math.floor(Math.random() * colors.length)] : '180,180,185'; } update() { this.x += this.vx; this.y += this.vy; const dx = this.x - mx, dy = this.y - my; const dist = Math.sqrt(dx*dx + dy*dy); if (dist < 150) { this.x += (dx / dist) * 0.5; this.y += (dy / dist) * 0.5; } if (this.x < 0) this.x = width; if (this.x > width) this.x = 0; if (this.y < 0) this.y = height; this.alpha += this.alphaSpeed; const maxA = this.isDiamond ? 1.0 : 0.55; const minA = this.isDiamond ? 0.1 : 0.03; if (this.alpha > maxA) { this.alpha = maxA; this.alphaSpeed *= -1; } if (this.alpha < minA) { this.alpha = minA; this.alphaSpeed *= -1; } } draw() { ctx.save(); ctx.translate(this.x, this.y); ctx.globalAlpha = this.alpha; if (this.isDiamond) { ctx.fillStyle = `rgba(${this.color}, 1)`; ctx.beginPath(); ctx.moveTo(0, -this.size * 3); ctx.quadraticCurveTo(this.size*0.2, -this.size*0.2, this.size * 3, 0); ctx.quadraticCurveTo(this.size*0.2, this.size*0.2, 0, this.size * 3); ctx.quadraticCurveTo(-this.size*0.2, this.size*0.2, -this.size * 3, 0); ctx.quadraticCurveTo(-this.size*0.2, -this.size*0.2, 0, -this.size * 3); ctx.fill(); } else { ctx.fillStyle = `rgba(${this.color}, 1)`; ctx.beginPath(); ctx.arc(0, 0, this.size, 0, Math.PI * 2); ctx.fill(); } ctx.restore(); } } class Comet { constructor() { this.x = Math.random() * (width * 0.5); this.y = Math.random() * (height * 0.3); this.speed = Math.random() * 15 + 15; this.angle = (Math.random() * 15 + 30) * Math.PI / 180; this.vx = Math.cos(this.angle) * this.speed; this.vy = Math.sin(this.angle) * this.speed; this.length = Math.random() * 150 + 120; this.active = true; } update() { this.x += this.vx; this.y += this.vy; if (this.x > width || this.y > height) this.active = false; } draw() { ctx.save(); const tailX = this.x - Math.cos(this.angle) * this.length; const tailY = this.y - Math.sin(this.angle) * this.length; const grad = ctx.createLinearGradient(this.x, this.y, tailX, tailY); grad.addColorStop(0, 'rgba(255, 255, 255, 1)'); grad.addColorStop(1, 'rgba(255, 255, 255, 0)'); ctx.strokeStyle = grad; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(this.x, this.y); ctx.lineTo(tailX, tailY); ctx.stroke(); ctx.fillStyle = 'rgba(255, 255, 255, 0.45)'; ctx.beginPath(); ctx.arc(this.x, this.y, 4, 0, Math.PI*2); ctx.fill(); ctx.fillStyle = 'rgba(255, 255, 255, 1)'; ctx.beginPath(); ctx.arc(this.x, this.y, 1.5, 0, Math.PI*2); ctx.fill(); ctx.restore(); } } for (let i = 0; i < 182; i++) stars.push(new Star()); function loop() { ctx.clearRect(0, 0, width, height); stars.forEach(s => { s.update(); s.draw(); }); if (Math.random() < 0.0025 && comets.length < 2) comets.push(new Comet()); for (let i = comets.length - 1; i >= 0; i--) { comets[i].update(); comets[i].draw(); if (!comets[i].active) comets.splice(i, 1); } requestAnimationFrame(loop); } loop();
